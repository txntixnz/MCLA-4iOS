cmake_minimum_required(VERSION 3.25)
project(MCLACPUCompilation LANGUAGES C CXX OBJC OBJCXX)

if(NOT APPLE OR NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    message(FATAL_ERROR "Use the iPhoneOS SDK and -DCMAKE_SYSTEM_NAME=iOS")
endif()
if(NOT CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "16.0" OR
   NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "This check must target arm64 / iOS 16.0")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 17)
option(MCLA_AOT_PROBE_ONLY "Compile public synthetic CPU probe, without game-derived inputs" OFF)
get_filename_component(MCLA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(MCLA_REXGLUE_SDK_DIR "${MCLA_ROOT}/theft4-foundation/glue/rexglue-sdk-main")
set(MCLA_PRIVATE_GENERATED_DIR "${MCLA_ROOT}/MCLAApp/generated" CACHE PATH "Private generated sources")

# Match the full app's foundation flags. This target has no graphics resources
# or launcher packaging, so missing shader metadata cannot mask CPU errors.
set(REXGLUE_CORE_ONLY ON CACHE BOOL "" FORCE)
set(REXGLUE_RUNTIME_ONLY ON CACHE BOOL "" FORCE)
set(REXGLUE_HEADLESS_KERNEL ON CACHE BOOL "" FORCE)
set(REXGLUE_HEADLESS_RENDER_AUDIO ON CACHE BOOL "" FORCE)
set(REXGLUE_HEADLESS_XMA_DECODER ON CACHE BOOL "" FORCE)
set(REXGLUE_HEADLESS_INPUT ON CACHE BOOL "" FORCE)
set(REXGLUE_USE_VULKAN OFF CACHE BOOL "" FORCE)
set(REXGLUE_USE_D3D12 OFF CACHE BOOL "" FORCE)
set(REXGLUE_ENABLE_TRACY OFF CACHE BOOL "" FORCE)
set(REXGLUE_ENABLE_PERF_COUNTERS OFF CACHE BOOL "" FORCE)
set(REXGLUE_BUILD_GTA4_RECOMP OFF CACHE BOOL "" FORCE)
set(REXGLUE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CMAKE_HAVE_LIBC_PTHREAD 1 CACHE INTERNAL "" FORCE)
set(CMAKE_USE_PTHREADS_INIT 1 CACHE INTERNAL "" FORCE)
set(CMAKE_THREAD_LIBS_INIT "" CACHE STRING "" FORCE)
add_subdirectory("${MCLA_REXGLUE_SDK_DIR}" rexglue)

if(MCLA_AOT_PROBE_ONLY)
    add_library(MCLACPUCompilation STATIC "${MCLA_ROOT}/tools/ios16_aot_toolchain_probe.cpp")
    message(STATUS "PUBLIC SYNTHETIC PROBE: actual game compilation is not being tested")
else()
    if(NOT EXISTS "${MCLA_PRIVATE_GENERATED_DIR}/sources.cmake")
        message(FATAL_ERROR "Stage private generated sources locally; never commit them to the public repository")
    endif()
    include("${MCLA_PRIVATE_GENERATED_DIR}/sources.cmake")
    add_library(MCLACPUCompilation STATIC ${GENERATED_SOURCES})
    target_include_directories(MCLACPUCompilation PUBLIC "${MCLA_PRIVATE_GENERATED_DIR}")
    set_source_files_properties("${MCLA_PRIVATE_GENERATED_DIR}/larecomp_register.cpp"
        PROPERTIES COMPILE_OPTIONS "-O0;-fno-global-isel")
endif()
target_compile_features(MCLACPUCompilation PUBLIC cxx_std_23)
target_compile_options(MCLACPUCompilation PRIVATE -fms-extensions)
target_link_libraries(MCLACPUCompilation PUBLIC rex::runtime)
